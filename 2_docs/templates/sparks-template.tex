\textbf{Python} --- inline use of \sparkscode{numpy} for a simple Bellman iteration:

\begin{lstlisting}[language=Python]
import numpy as np

def solve_bellman(params, tol=1e-6, max_iter=1000):
    """Value-function iteration for a discrete state space."""
    V = np.zeros(params.n_states)
    for it in range(max_iter):
        V_new, policy = bellman_operator(V, params)
        if np.max(np.abs(V_new - V)) < tol:
            return V_new, policy, it
        V = V_new
    return V, policy, max_iter
\end{lstlisting}
